    ```python
    def gate_commands(conv, only=None):
        """``(cmds, env)``: every command the gate runs for ``conv``, in the order it runs them,
        and the environment assignments lifted out of the test command's leading ``NAME=value``
        tokens (:func:`command_env`).

        The product's test command first, then — only when this is not a bisection's targeted
        re-run (``only``) — its :func:`gate_checks`. This is the one list: :func:`product_gate`
        runs it and :mod:`asf.workers.worker_perms` permits it, so a command appended to a
        product's ``conventions.gate_checks`` is allowed in every worker session with no second
        edit.
        """
        cmds, extra = [], {}
        if conv.test_command:
            cmd, extra = command_env(shlex.split(str(conv.test_command)))
            cmds.append(cmd)
        if not only:
            cmds += gate_checks(conv)
        return cmds, extra
    ```

**Files**

- `asf/workers/worker_perms.py` — new.
  ```python
  """asf.workers.worker_perms — F-0115: the rules a worker session is launched with.

  A worker session has nobody to answer an approval prompt: a Bash call with no allow rule is
  refused, recorded in the result's ``permission_denials``, and reported by the session as a check
  "left out" — so its branch lands with the repo's own lint unverified. The rules are therefore
  derived, never kept by hand: :func:`gate_allow_rules` reads
  :func:`asf.harvest.harvest.gate_commands`, the same list the gate itself runs for the product —
  its test command and its own ``conventions.gate_checks`` — so a command appended to a product's
  gate is allowed the day it is added.

  :func:`settings` merges those rules **over** the operator's own ``worker_pool.settings_file`` —
  `asf.console_perms.merge`'s discipline: append what is missing, keep every existing entry and
  its order, touch no other key. The operator's file is never written to and no rule of theirs is
  ever removed; where one of theirs denies what this module allows, the runtime's deny wins and
  :func:`check_doctor` says so by name.
  """
  import json
  import os
  import shlex

  from asf import approvals, env

  #: What a session needs to read its own situation, and nothing that can run a program.
  #: ``printenv`` prints variables and cannot execute (``env NAME=v prog`` can, and is not here);
  #: ``grep`` has been on :data:`asf.approvals._READ_ONLY_PROGRAMS` since F-0042, and the card's
  #: own evidence is ``printenv | grep`` — the runtime judges each side of a pipe on its own.
  SELF_DIAGNOSIS = ('Bash(printenv:*)', 'Bash(grep:*)')


  def rule_for(argv):
      """One ``Bash(<command>:*)`` rule for ``argv`` — ``shlex.join``'s spelling, so the rule
      reads exactly as the gate invokes it (C5); ``None`` for an empty ``argv``."""


  def gate_allow_rules(product):
      """One rule per command the gate runs for ``product`` (:func:`harvest.gate_commands` over
      ``product.conventions``), plus ``conventions.pre_push_check`` — the command every code
      brief names as the one that must pass before a push. A command
      ``asf.approvals.full_suite_command`` matches is left out: the approvals hook refuses it
      whatever the runtime allows, and two fences on one command read as one flaky fence (C4).
      Order: the gate's own, then ``pre_push_check``; each rule once."""


  def allow_rules(product):
      """:func:`gate_allow_rules` then :data:`SELF_DIAGNOSIS`, deduped, order kept."""


  def settings(product, operator=None):
      """``operator`` (the object read from ``worker_pool.settings_file``, or ``{}``) with every
      rule of :func:`allow_rules` present exactly once in ``permissions.allow`` — every unrelated
      key, and every deny rule, untouched. A ``permissions`` block with no ``deny`` key comes back
      with no ``deny`` key."""


  def path(product, wp):
      """``<state_dir>/worker-settings.json``, rewritten only when it differs from what is there;
      the path :func:`asf.workers.spawn.spawn` gives ``Job(settings_file=…)``. ``wp`` is the
      worker pool block: its ``settings_file`` is read through
      :func:`asf.workers.spawn.settings_file`, so a configured path that does not exist still
      refuses the launch."""


  def denied_anyway(product, wp):
      """Every rule of :func:`allow_rules` that the composed object's ``permissions.deny`` also
      matches — in the runtime a deny beats an allow, and a rule in both refuses (C9). Composes
      ``settings`` in memory; never calls :func:`path`, never writes a file."""


  def check_doctor(cfg, product):
      """``(ok, detail)`` — the ``worker permissions`` doctor row. ``(True, 'no worker pool')``
      for a config with no worker pool or no accounts; otherwise green with the count derived,
      red naming every rule :func:`denied_anyway` finds and the configuration key its deny came
      from. A configured-but-missing operator settings file is a red row naming the missing path,
      never a traceback."""


  def offer_text(product, wp=None):
      """Every rule the next session will get, in full, one per line, the gate's own marked with
      the command they came from — `console_perms.offer_text`'s discipline: a confirmation of a
      list has to be of the actual list. No ``install`` counterpart (C10): the worker's list is
      composed by ASF at every launch, so there is nothing for an operator to install."""
  ```
  Three notes on what is deliberately not there, unchanged from the spec: no `FIXED_ALLOW` (every
  rule is derived or named for a stated reason); no `deny_rules` (F-0062's); no `tools/` literal
  and no `.claude/` literal anywhere in the module, docstring included (P11 — the gate's script
  paths arrive already-split out of `gate_checks(conv)`'s own config read, and the composed file
